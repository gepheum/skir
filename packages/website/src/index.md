    Point::whole{.x = 0.0f, .y = 0.0f},

    skir::Parse<Shape>(skir::ToDenseJson(triangle));

    skir::service::MakeHttplibClient(
        std::make_unique<httplib::Client>(
            "localhost",
            8080,
        ),
        "/api",
    );

    InvokeRemote(*client, GetArea(), triangle);
std::cout << *area << "\n";  // 50
